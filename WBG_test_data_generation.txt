text_splitter = TokenTextSplitter(chunk_size=1024, chunk_overlap=50)
# Regex to match illegal control characters
ILLEGAL_CHARACTERS_RE = re.compile(r'[\x00-\x08\x0b-\x0c\x0e-\x1f]')

def clean_text(value):
    """Remove illegal Excel characters from a string."""
    if isinstance(value, str):
        return ILLEGAL_CHARACTERS_RE.sub("", value)
    return value

# extracting text from files
def extract_text_from_txts(directory):
    abstracts = []
    doc_to_text_mapping = {}
    files = [f for f in os.listdir(directory) if f.endswith((".txt", ".pdf", ".doc", ".docx"))]

    for file in files:
        file_path = os.path.join(directory, file)
        text = ""

        if file.endswith(".txt"):
            with open(file_path, "r", encoding="utf-8", errors="ignore") as f:
                text = f.read()
        elif file.endswith(".pdf"):
            try:
                doc = fitz.open(file_path)
                for page in doc:
                    text += page.get_text()
                doc.close()
            except Exception as e:
                print(f"Error reading PDF {file}: {e}")
                continue
        elif file.endswith(".docx"):

                doc = docx.Document(file_path)
                text="\n".join([para.text for para in doc.paragraphs if para.text.strip()])

# using text splitter for making chunks
        if text.strip():
            chunks = text_splitter.split_text(text)
            abstracts.extend(chunks)
            doc_to_text_mapping.update({chunk: file for chunk in chunks})

    return abstracts, doc_to_text_mapping

def generate_multi_turn_conversation(goal, context, persona="Product Owner"):

            user_msg = goal
            conversation.append({"user": user_msg})
        else:
            history = "\n".join(
                [f"User: {m['user']}" if 'user' in m else f"Bot: {m['bot']}"
                 for m in conversation]
            )

Continue this multi-turn conversation.

Persona: {persona}
Goal: {goal}
Relevant context:
{context}

Conversation so far:
{history}

What would the user ask next? Respond ONLY with the next user message.
""".strip()

        # Next user message

        # ✅ Stop if goal achieved

    return conversation
